## 从接口抓取数据

从普通的没有任何安全的措施的接口中获取数据

url:"https://spa1.scrape.center/"

~~~py
import logging
import requests as r
import time
logging.basicConfig(
    level=logging.INFO,
    format="%(asctime)s - %(name)s - %(levelname)s - %(message)s",
)

def api_spider():
    url = "https://spa1.scrape.center/api/movie/"
    params = {
        "limit":10,
        "offset":0
    }
    i = 0
    with open("spider_.txt", "a", encoding="utf-8") as file:
        while True:
            response:dict = r.get(url="https://spa1.scrape.center/api/movie/",params=params).json()
            result = response.get('results',{})
            if not result :
                break
            total:int = response.get('count')
            if i >= total:
                break
            for item in result:
                id = item.get('id')
                details = r.get(url=f"{url}{id}").json()
                file.write(f"ID: {details.get('id')}\n")
                file.write(f"名称: {details.get('name')}\n")
                file.write(f"详情: {details.get('drama')}\n")
                file.write(f"创建时间: {details.get('published_at')}\n")
                file.write(f"评分: {details.get('score')}\n")
                file.write("-" * 30 + "\n")
            i = i + 10
            params['offset'] = i
            time.sleep(1)



if __name__ == "__main__":
    api_spider()
~~~

### 文件目录
```python
water_tenders/ # Scrapy 项目包
    items.py # 定义要抓取的数据结构，例如公告标题、发布时间、来源网站、详情链接和正文。
    middlewares.py # 请求和响应处理中间件，例如请求头、代理、重试和 Cookie。
    pipelines.py # 数据处理管道，用于去重、清洗、写入 Excel 或数据库。
    settings.py # Scrapy 全局设置，例如并发数、下载延迟和 robots.txt 策略。
    spiders/ # 存放爬虫文件，每个网站通常对应一个 spider。
    __init__.py
scrapy.cfg # Scrapy 的配置入口。告诉 Scrapy 默认设置文件在哪里
```

        ~~~py
        import scrapy
        from water_tenders.items import QuoteItem          

        class ToscrapeSpider(scrapy.Spider):
            name = "toscrape"
            allowed_domains = ["quotes.toscrape.com"]
            start_urls = ["https://quotes.toscrape.com/"]

            def parse(self, response):
                for quote_item in response.css("div.quote"):
                    item = QuoteItem()
                    item.author = quote_item.css("small.author::text").extract_first()
                    item.tags = quote_item.css("div.tags a.tag::text").extract()
                    item.text = quote_item.css("div.quote span.text::text").extract_first()
                    yield item # 暂停 丢给引擎 引擎将item丢给pipeline 处理之后继续下一步
                next = response.css('nav ul.pager li.next a::attr("href")').extract_first() # 拿到分页路径
                url = response.urljoin(next) # 和原始url拼接成新的url
                yield scrapy.Request(url, callback=self.parse) # 暂停 丢给引擎  引擎生成新的请求 
        ~~~
