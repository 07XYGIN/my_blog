## Scrapy

## 创建项目

```bash
uv run scrapy startproject water_tenders .
```

## 常用命令

```bash
uv run scrapy genspider ggzyfw.beijing ggzyfw.beijing.gov.cn # 新建一个 spider，文件名为 ggzyfw.beijing，对应 URL 为 ggzyfw.beijing.gov.cn

uv run scrapy list # 查询项目中有哪些 spider

uv run scrapy crawl genspider # 运行指定 spider

uv run scrapy shell "https://example.com" # 调试网页
```

### 文件目录
```python
water_tenders/ # 真正的 Scrapy 项目包
  items.py # 定义要抓取的数据结构，例如公告标题、发布时间、来源网站、详情链接和正文。
  middlewares.py # 请求和响应处理中间件，例如请求头、代理、重试和 Cookie。
  pipelines.py # 数据处理管道，用于去重、清洗、写入 Excel 或数据库。
  settings.py # Scrapy 全局设置，例如并发数、下载延迟和 robots.txt 策略。
  spiders/ # 存放爬虫文件，每个网站通常对应一个 spider。
    __init__.py
scrapy.cfg # Scrapy 的配置入口。告诉 Scrapy 默认设置文件在哪里
```


### settings.py


```python
# Scrapy settings for water_tenders project

BOT_NAME = "water_tenders"
SPIDER_MODULES = ["water_tenders.spiders"]
NEWSPIDER_MODULE = "water_tenders.spiders"

# ========== 请求与反爬 ==========
USER_AGENT = "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120.0.0.0 Safari/537.36"
ROBOTSTXT_OBEY = False  # 很多网站robots.txt限制严格，爬取公开数据可关闭
DOWNLOAD_DELAY = 2       # 请求间隔2秒
CONCURRENT_REQUESTS_PER_DOMAIN = 2  # 每域名并发数
RANDOMIZE_DOWNLOAD_DELAY = True  # 在延迟基础上随机±0.5秒，更像真人

# ========== 请求头 ==========
DEFAULT_REQUEST_HEADERS = {
    "Accept": "text/html,application/xhtml+xml,application/xml;q=0.9,image/webp,*/*;q=0.8",
    "Accept-Language": "zh-CN,zh;q=0.9,en;q=0.8",
    "Accept-Encoding": "gzip, deflate, br",
    "Connection": "keep-alive",
    "Upgrade-Insecure-Requests": "1",
}

# ========== 下载中间件（常用的几个） ==========
DOWNLOADER_MIDDLEWARES = {
    "scrapy.downloadermiddlewares.useragent.UserAgentMiddleware": None,  # 禁用默认UA
    "scrapy.downloadermiddlewares.retry.RetryMiddleware": 90,
    "scrapy.downloadermiddlewares.httpproxy.HttpProxyMiddleware": 110,
}

# ========== 重试机制 ==========
RETRY_ENABLED = True
RETRY_TIMES = 3  # 失败重试3次
RETRY_HTTP_CODES = [500, 502, 503, 504, 408, 429]  # 这些状态码触发重试

# ========== 重定向 ==========
REDIRECT_ENABLED = True
REDIRECT_MAX_TIMES = 3  # 最多跟随3次重定向

# ========== 超时设置 ==========
DOWNLOAD_TIMEOUT = 30  # 下载超时30秒

# ========== 数据管道 ==========
ITEM_PIPELINES = {
    # "water_tenders.pipelines.WaterTendersPipeline": 300,  # 取消注释启用
}

# ========== 自动限速（推荐） ==========
AUTOTHROTTLE_ENABLED = True
AUTOTHROTTLE_START_DELAY = 1.0
AUTOTHROTTLE_MAX_DELAY = 10.0
AUTOTHROTTLE_TARGET_CONCURRENCY = 1.0
AUTOTHROTTLE_DEBUG = False

# ========== 日志 ==========
LOG_LEVEL = "INFO"
LOG_FILE = "logs/scrapy.log"
LOG_FILE_APPEND = True
LOG_FORMAT = "%(asctime)s [%(name)s] %(levelname)s: %(message)s"

# ========== 输出编码 ==========
FEED_EXPORT_ENCODING = "utf-8"

# ========== 内存/性能 ==========
MEMUSAGE_ENABLED = True  # 监控内存使用
MEMUSAGE_LIMIT_MB = 512  # 内存超过512MB时警告
CLOSESPIDER_TIMEOUT = 3600  # 爬虫运行1小时后自动关闭（防止死循环）

# ========== 调试（开发时启用） ==========
# HTTPCACHE_ENABLED = True  # 开发时启用缓存，避免重复请求
# HTTPCACHE_EXPIRATION_SECS = 86400  # 缓存24小时
# HTTPCACHE_DIR = "httpcache"
# HTTPCACHE_IGNORE_HTTP_CODES = [500, 502, 503]
```
