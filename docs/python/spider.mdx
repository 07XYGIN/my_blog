## 从接口抓取数据

从普通的没有任何安全的措施的接口中获取数据

url:"https://spa1.scrape.center/"

~~~py
import logging
import requests as r
import time
logging.basicConfig(
    level=logging.INFO,
    format="%(asctime)s - %(name)s - %(levelname)s - %(message)s",
)

def api_spider():
    url = "https://spa1.scrape.center/api/movie/"
    params = {
        "limit":10,
        "offset":0
    }
    i = 0
    with open("spider_.txt", "a", encoding="utf-8") as file:
        while True:
            response:dict = r.get(url="https://spa1.scrape.center/api/movie/",params=params).json()
            result = response.get('results',{})
            if not result :
                break
            total:int = response.get('count')
            if i >= total:
                break
            for item in result:
                id = item.get('id')
                details = r.get(url=f"{url}{id}").json()
                file.write(f"ID: {details.get('id')}\n")
                file.write(f"名称: {details.get('name')}\n")
                file.write(f"详情: {details.get('drama')}\n")
                file.write(f"创建时间: {details.get('published_at')}\n")
                file.write(f"评分: {details.get('score')}\n")
                file.write("-" * 30 + "\n")
            i = i + 10
            params['offset'] = i
            time.sleep(1)



if __name__ == "__main__":
    api_spider()
~~~

def ssr_spider():
    baseurl = "https://login2.scrape.center/login"
    body = {
        "username":"admin",
        "password":"admin"
    }
    result = r.post(url = baseurl,data = body,allow_redirects=False) # 拿到原始响应

    cookies = result.cookies
    data = r.get("https://login2.scrape.center/",cookies=cookies) # 拿到html
    soup = BeautifulSoup(data.text, 'html.parser')

    movie_cards = soup.select('.el-card.item') # 使用bs4分析DOM结构
    for card in movie_cards:
        detail_link = card.select_one('.name')['href']

            region = ""

## Scrapy

### 创建项目

```bash
uv run scrapy startproject water_tenders .
```

### 常用命令

```bash
uv run scrapy genspider ggzyfw.beijing ggzyfw.beijing.gov.cn # 新建一个 spider，文件名为 ggzyfw.beijing，对应 URL 为 ggzyfw.beijing.gov.cn

uv run scrapy list # 查询项目中有哪些 spider

uv run scrapy crawl genspider # 运行指定 spider

uv run scrapy shell "https://example.com" # 调试网页
```

### 文件目录
```python
water_tenders/ # 真正的 Scrapy 项目包
    items.py # 定义要抓取的数据结构，例如公告标题、发布时间、来源网站、详情链接和正文。
    middlewares.py # 请求和响应处理中间件，例如请求头、代理、重试和 Cookie。
    pipelines.py # 数据处理管道，用于去重、清洗、写入 Excel 或数据库。
    settings.py # Scrapy 全局设置，例如并发数、下载延迟和 robots.txt 策略。
    spiders/ # 存放爬虫文件，每个网站通常对应一个 spider。
    __init__.py
scrapy.cfg # Scrapy 的配置入口。告诉 Scrapy 默认设置文件在哪里
```
